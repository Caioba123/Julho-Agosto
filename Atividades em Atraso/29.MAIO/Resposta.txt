class Produto {
  String _nome;
  double _preco;
  int _quantidade;

  Produto(this._nome, this._preco, this._quantidade) {
    if (_preco <= 0) {
      _preco = 1; // valor mínimo válido
    }
    if (_quantidade < 0) {
      _quantidade = 0;
    }
  }

  // Getters
  String get nome => _nome;
  double get preco => _preco;
  int get quantidade => _quantidade;

  // Setters com validação
  set preco(double valor) {
    if (valor > 0) _preco = valor;
  }

  set quantidade(int valor) {
    if (valor >= 0) _quantidade = valor;
  }

  void mostrarDados() {
    print("Produto: $_nome | Preço: R\$ $_preco | Quantidade: $_quantidade");
  }

  void calcularTotal() {
    print("Total em estoque: R\$ ${_preco * _quantidade}");
  }
}

void main() {
  Produto p1 = Produto("Mouse", 50, 10);
  Produto p2 = Produto("Teclado", -120, 5); // inválido → ajustado
  Produto p3 = Produto("Monitor", 900, -3); // inválido → ajustado

  p1.mostrarDados();
  p1.calcularTotal();

  p2.mostrarDados();
  p2.calcularTotal();

  p3.mostrarDados();
  p3.calcularTotal();
}

  if (removido) {
    print("Produto $remover removido.");

class Aluno {
  String _nome;
  double _nota;

  Aluno(this._nome, this._nota) {
    if (_nota < 0 || _nota > 10) {
      _nota = 0; // valor inválido ajustado
    }
  }

  String get nome => _nome;
  double get nota => _nota;

  set nota(double valor) {
    if (valor >= 0 && valor <= 10) _nota = valor;
  }

  String verificarSituacao() {
    if (_nota >= 7) return "Aprovado";
    if (_nota >= 5) return "Recuperação";
    return "Reprovado";
  }

  void mostrarDados() {
    print("Aluno: $_nome | Nota: $_nota | Situação: ${verificarSituacao()}");
  }
}

void main() {
  Aluno a1 = Aluno("Gabriel", 8);
  Aluno a2 = Aluno("Luiz", 6);
  Aluno a3 = Aluno("Pedro", 12); // inválido → ajustado

  a1.mostrarDados();
  a2.mostrarDados();
  a3.mostrarDados();
}

  No? direito;

class Arvore {
  No? raiz;

  void inserir(int valor) {
    raiz = _inserirRec(raiz, valor);
  }

  No _inserirRec(No? no, int valor) {
    if (no == null) return No(valor);

    if (valor < no.valor) {
      no.esquerdo = _inserirRec(no.esquerdo, valor);
    } else {
      no.direito = _inserirRec(no.direito, valor);
    }
    return no;
  }
}

void main() {

  arvore.inserir(20);
  arvore.inserir(40);
  arvore.inserir(60);

}
